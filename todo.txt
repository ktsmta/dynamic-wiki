@event.listens_for(Category, "after_insert")
def create_wiki_for_category(mapper, connection, target):
    # target は作成された Category レコード
    connection.execute(
        Wiki.__table__.insert().values(
            category_id=target.id,
            title=target.name,   # デフォルトでカテゴリ名をタイトルにする例
            content=""           # 初期内容は空
        )
    )
